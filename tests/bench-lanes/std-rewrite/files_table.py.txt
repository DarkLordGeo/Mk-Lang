std-rewrite's write set, with the rewrite bar's searchable rules checked per
file and the async-bar reading status from a hand-kept map (READ below), and
regenerates lines-after.txt, the after line count of every file (`wc -l`:
the newline count), from the same walk, so no row is patched by hand
(a hand-patched row drifts from the tree).
Run from the repository root with python -I. With --check it writes nothing: it compares
every row of lines-after.txt with the tree and exits 1 on any mismatch, a
missing row or a row whose file is gone; run it before relying on the counts
(std/std-rewrite_rows_test.go's TestSRW_LinesAfterMatchesTree does the
same in Go when MK_SRW_LINES=1)."""

        "std/hash", "std/mem", "std/cli", "std/log"]

def counts(path):
    d = {}
    if not os.path.exists(path):
        return d
    for line in open(path, encoding="utf-8"):
        m = re.match(r"\s*(\d+)\s+(std/\S+\.mk)$", line)
        if m:
            d[m.group(2)] = int(m.group(1))
    return d

after_old = counts("tests/bench-lanes/std-rewrite/lines-after.txt")

SKIPWORDS = ("StartsWith(", "EndsWith(", "ScopeWith(")

rows = []
for d in DIRS:
    for dp, _, fs in os.walk(d):
        if ".mk-cache" in dp:
            continue
        for f in sorted(fs):
            if not f.endswith(".mk"):
                continue
            p = os.path.join(dp, f).replace("\\", "/")
            text = open(p, encoding="utf-8").read()
            n = text.count("\n")
            r1 = len(re.findall(r"const [a-z_]+\s*:\s*fn\(", text))
            r2 = sum(1 for m in re.finditer(r"^export [\w\[\], .]*?(\w+With)\(", text, re.M) if not m.group(1) + "(" in SKIPWORDS)
            r3 = text.count('", ())')
            conc = bool(re.search(r"raw\.Atomic|rt\.Lock|\.Cas\(|WaitWord|IoWait|Park\(", text))
            test = os.path.basename(p).startswith("tests_")
            rows.append((p, before.get(p), n, r1, r2, r3, conc, test, READ.get(p, "")))

rows.sort()

AFTER = "tests/bench-lanes/std-rewrite/lines-after.txt"
if CHECK:
    now = {r[0]: r[2] for r in rows}
    bad = []
    for p, n in sorted(after_old.items()):
        if p not in now:
            bad.append("%s: in lines-after.txt (%d), not in the tree" % (p, n))
        elif now[p] != n:
            bad.append("%s: lines-after.txt %d, tree %d" % (p, n, now[p]))
    for p in sorted(set(now) - set(after_old)):
        bad.append("%s: in the tree (%d), no row in lines-after.txt" % (p, now[p]))
    for b in bad:
        print(b)
    print("lines-after.txt: %d rows, %d mismatches" % (len(after_old), len(bad)))
    sys.exit(1 if bad else 0)

      "it with the tree. Last generated by the rerun 2 fix round, 2026-11-08; earlier versions are in git "
      "history; compare lines-before.txt, same command"]
for r in rows:
    la.append("%7d %s" % (r[2], r[0]))
open(AFTER, "w", encoding="utf-8", newline="\n").write("\n".join(la) + "\n")

Columns: lines before (lines-before.txt, 65ba5ca; blank: the file is newer) and now (`wc -l`); R1 `const` fn parameters declared (rule 1 can apply only at their call sites; every call site in std was checked in rerun session 2: none carries a capture-only context struct); R2 exported `...With(` declarations other than StartsWith, EndsWith and ScopeWith (DD-2: 0 everywhere; internal helpers such as std/sys's `FailWith` are not DD-2's forms); R3 `\", ())` (the short Print form; the 2 in tests_fmt.mk test the explicit empty record on purpose); C: atomics, locks or waits in the file (the async bar applies); read: who read the file line by line under the async bar (blank: not yet).")

    tot_a += a

out.append("")

open("tests/bench-lanes/std-rewrite/files.md", "w", encoding="utf-8", newline="\n").write("\n".join(out) + "\n")
